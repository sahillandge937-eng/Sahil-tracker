const express = require('express'); const path = require('path'); const app = express(); app.use(express.json()); // Serve static frontend files from 'public' directory app.use(express.static(path.join(__dirname, '../public'))); // Fallback route to serve index.html for unknown non-API routes app.get('*', (req, res, next) => { if (req.path.startsWith('/api')) { return next(); } res.sendFile(path.join(__dirname, '../public/index.html')); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
